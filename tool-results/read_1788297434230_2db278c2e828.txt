   132→class WesternNatalRequest(BaseModel):
   133→    """Request schema for Western (tropical) natal chart calculation."""
   134→    birth_datetime: datetime = Field(
   135→        ...,
   136→        description="Birth datetime with timezone in ISO 8601 format (e.g., '1990-01-15T14:30:00+06:30'). Must include timezone for accurate calculations.",
   137→        examples=["1990-01-15T14:30:00+06:30"]
   138→    )
   139→    latitude: float = Field(...,
   140→        ge=-90,
   141→        le=90,
   142→        description="Geographic latitude of birth location in decimal degrees. North is positive, South is negative.", json_schema_extra={"example": 16.8661})
   143→    longitude: float = Field(...,
   144→        ge=-180,
   145→        le=180,
   146→        description="Geographic longitude of birth location in decimal degrees. East is positive, West is negative.", json_schema_extra={"example": 96.1951})
   147→    house_system: Literal['placidus', 'koch', 'equal', 'whole_sign'] = Field(default='placidus',
   148→        description="House system to use for house cusp calculations.")
   149→
   150→    @field_validator('birth_datetime')
   151→    @classmethod
   152→    def validate_birth_datetime(cls, v):
   153→        """Validate birth datetime is not in future and within reasonable range."""
   154→        if v > datetime.now(v.tzinfo):
   155→            raise ValueError("Birth datetime cannot be in the future")
   156→        if v.year < 1900:
   157→            raise ValueError(
   158→                "Birth year must be 1900 or later for accurate calculations")
   159→        if v.year > datetime.now().year:
   160→            raise ValueError("Birth year cannot be in the future")
   161→        return v
   162→
   163→    @field_validator('latitude', 'longitude')
   164→    @classmethod
   165→    def validate_coordinates(cls, v):
   166→        """Validate coordinates are numeric and within valid range."""
   167→        if not isinstance(v, (int, float)):
   168→            raise ValueError("Coordinates must be numeric values")
   169→        if not (-180 <= v <= 180):
   170→            raise ValueError("Coordinate value out of valid range")
   171→        return v

   364→class MahaboteCalculationRequest(BaseModel):
   365→    """Request schema for Mahabote (Myanmar traditional astrology) calculation."""
   366→    birth_date: Date = Field(..., description="Birth date")
   367→    birth_time: Optional[datetime] = Field(
   368→        None,
   369→        description="Optional birth time for Rahu rule (Wednesday afternoon). Accepts ISO 8601 format with timezone offset (e.g., '1987-12-04T03:40:00+06:30')"
   370→    )
   371→
   372→    @field_validator('birth_date')
   373→    @classmethod
   374→    def validate_birth_date(cls, v):
   375→        """Validate birth date is not in future."""
   376→        if v > Date.today():
   377→            raise ValueError("Birth date cannot be in the future")
   378→        if v.year < 1900:
   379→            raise ValueError("Birth year must be 1900 or later")
   380→        return v
   381→
   382→    @field_validator('birth_time', mode='before')
   383→    @classmethod
   384→    def parse_birth_time(cls, v):
   385→        """Parse birth time from ISO 8601 string or datetime object."""
   386→        if v is None:
   387→            return None
   388→        if isinstance(v, datetime):
   389→            return v
   390→        if isinstance(v, str):
   391→            try:
   392→                import re
   393→                match = re.match(r'(.+)([+-]\d{2}):(\d{2})$', v)
   394→                if match:
   395→                    v = f"{match.group(1)}{match.group(2)}{match.group(3)}"
   396→                return datetime.fromisoformat(v)
   397→            except ValueError as e:
   398→                raise ValueError(f"Invalid datetime format: {str(e)}")
   399→        raise ValueError(f"birth_time must be a datetime object or ISO 8601 string, got {type(v)}")
   400→

   586→    @classmethod
   587→    def parse_birth_time(cls, v):
   588→        """Parse birth time from various formats (HH:MM, HH:MM:SS, or ISO datetime)."""
   589→        if v is None:
   590→            return None
   591→        if isinstance(v, str):
   592→            if 'T' in v:
   593→                time_part = v.split('T')[1]
   594→                if '+' in time_part:
   595→                    time_part = time_part.split('+')[0]
   596→                if '-' in time_part and len(time_part) > 5:
   597→                    time_part = time_part.split('-')[0]
   598→                return time_part
   599→            return v
   600→        raise ValueError(f"birth_time must be a string, got {type(v)}")
